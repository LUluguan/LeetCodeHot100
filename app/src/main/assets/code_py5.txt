###0070
# 思路：斐波那契递推：f(n) = f(n-1) + f(n-2)，滚动变量 O(1) 空间。
class Solution:
    def climbStairs(self, n: int) -> int:
        prev, cur = 1, 1                 # f(0) = f(1) = 1
        for _ in range(2, n + 1):
            prev, cur = cur, prev + cur
        return cur
# 时间复杂度 O(n)，空间 O(1)。

###0118
# 思路：逐行构造：每行首尾 1，中间 = 上一行相邻两数之和。
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        res = []
        for i in range(numRows):
            row = [1] * (i + 1)
            for j in range(1, i):        # 中间元素
                row[j] = res[i - 1][j - 1] + res[i - 1][j]
            res.append(row)
        return res
# 时间复杂度 O(n^2)，空间 O(1)（不计输出）。

###0198
# 思路：打家劫舍 DP：f(i) = max(f(i-1) 不偷, f(i-2) + nums[i] 偷)。
class Solution:
    def rob(self, nums: List[int]) -> int:
        prev2 = prev1 = 0                # f(i-2), f(i-1)
        for x in nums:
            prev2, prev1 = prev1, max(prev1, prev2 + x)
        return prev1
# 时间复杂度 O(n)，空间 O(1)。

###0279
# 思路：完全背包：f(i) = min(f(i - j*j) + 1)，j*j <= i。
class Solution:
    def numSquares(self, n: int) -> int:
        dp = [0] + [float('inf')] * n
        for i in range(1, n + 1):
            j = 1
            while j * j <= i:            # 枚举最后一个平方数
                dp[i] = min(dp[i], dp[i - j * j] + 1)
                j += 1
        return dp[n]
# 时间复杂度 O(n·sqrt(n))，空间 O(n)。

###0322
# 思路：完全背包最少硬币数：f(a) = min(f(a - coin) + 1)，
# f(0)=0、其余无穷；结果仍为无穷表示凑不出。
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        dp = [0] + [float('inf')] * amount
        for a in range(1, amount + 1):
            for coin in coins:
                if coin <= a:
                    dp[a] = min(dp[a], dp[a - coin] + 1)
        return dp[amount] if dp[amount] != float('inf') else -1
# 时间复杂度 O(amount × 硬币数)，空间 O(amount)。

###0139
# 思路：DP：dp[i] 表示 s 前 i 个字符能否被拆分；枚举"最后一个单词"。
class Solution:
    def wordBreak(self, s: str, wordDict: List[str]) -> bool:
        words = set(wordDict)            # O(1) 查询
        n = len(s)
        dp = [False] * (n + 1)
        dp[0] = True                     # 空串可拆分
        for i in range(1, n + 1):
            for j in range(i):
                if dp[j] and s[j:i] in words:   # 前 j 个可拆 且 s[j..i) 是单词
                    dp[i] = True
                    break
        return dp[n]
# 时间复杂度 O(n^2)（含子串哈希），空间 O(n)。

###0300
# 思路：贪心 + 二分：tails[i] = 长度 i+1 的递增子序列的最小尾元素，
# 每个数二分它应放的位置（第一个 >= 它的槽位），O(n log n)。
import bisect

class Solution:
    def lengthOfLIS(self, nums: List[int]) -> int:
        tails = []                       # 有序数组
        for x in nums:
            pos = bisect.bisect_left(tails, x)   # 严格递增用 bisect_left
            if pos == len(tails):
                tails.append(x)          # 延长
            else:
                tails[pos] = x           # 替换，让尾部更小
        return len(tails)
# 时间复杂度 O(n log n)，空间 O(n)。

###0152
# 思路：同时维护以 i 结尾的最大/最小乘积（负数会翻转极值）。
class Solution:
    def maxProduct(self, nums: List[int]) -> int:
        fmax = fmin = best = nums[0]
        for x in nums[1:]:
            candidates = (x, fmax * x, fmin * x)   # 三选一：另起/接大/接小
            fmax, fmin = max(candidates), min(candidates)
            best = max(best, fmax)
        return best
# 时间复杂度 O(n)，空间 O(1)。

###0416
# 思路：01 背包可行性：目标 sum/2；dp[t] = 能否凑出和 t，
# 每个数只用一次，t 从大到小遍历。
class Solution:
    def canPartition(self, nums: List[int]) -> bool:
        total = sum(nums)
        if total % 2 != 0:               # 总和为奇数，不可能平分
            return False
        target = total // 2
        dp = [False] * (target + 1)
        dp[0] = True
        for x in nums:
            for t in range(target, x - 1, -1):   # 倒序：每个数只用一次
                dp[t] = dp[t] or dp[t - x]
            if dp[target]:               # 提前结束
                return True
        return dp[target]
# 时间复杂度 O(n·target)，空间 O(target)。

###0032
# 思路：栈存"最后一个未匹配位置"（哨兵）：')' 弹栈后，
# 栈空则当前下标成为新哨兵，否则长度 = i - 新栈顶。
class Solution:
    def longestValidParentheses(self, s: str) -> int:
        stack = [-1]                     # 哨兵
        best = 0
        for i, c in enumerate(s):
            if c == '(':
                stack.append(i)
            else:
                stack.pop()              # 弹出 '(' 或旧哨兵
                if not stack:
                    stack.append(i)      # 无配对，成为新哨兵
                else:
                    best = max(best, i - stack[-1])
        return best
# 时间复杂度 O(n)，空间 O(n)。

###0062
# 思路：一维滚动 DP：dp[j] += dp[j-1]（旧 dp[j] 是上方，dp[j-1] 是左方）。
class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        dp = [1] * n                     # 第一行全为 1
        for _ in range(1, m):
            for j in range(1, n):
                dp[j] += dp[j - 1]
        return dp[n - 1]
# 时间复杂度 O(mn)，空间 O(n)。

###0064
# 思路：一维滚动 DP：f(i,j) = grid[i][j] + min(上, 左)；
# 每行开始先更新 dp[0]。
class Solution:
    def minPathSum(self, grid: List[List[int]]) -> int:
        m, n = len(grid), len(grid[0])
        dp = [0] * n
        dp[0] = grid[0][0]
        for j in range(1, n):            # 第一行
            dp[j] = dp[j - 1] + grid[0][j]
        for i in range(1, m):
            dp[0] += grid[i][0]          # 第一列只能从上来
            for j in range(1, n):
                dp[j] = min(dp[j], dp[j - 1]) + grid[i][j]
        return dp[n - 1]
# 时间复杂度 O(mn)，空间 O(n)。

###1143
# 思路：LCS 二维 DP：尾字符相等取对角线+1，否则取两侧最优。
class Solution:
    def longestCommonSubsequence(self, text1: str, text2: str) -> int:
        m, n = len(text1), len(text2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if text1[i - 1] == text2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1] + 1
                else:
                    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
        return dp[m][n]
# 时间复杂度 O(mn)，空间 O(mn)。

###0072
# 思路：编辑距离 DP：尾字符相同无操作；
# 否则 = min(替换, 删除, 插入) + 1。边界 f(i,0)=i、f(0,j)=j。
class Solution:
    def minDistance(self, word1: str, word2: str) -> int:
        m, n = len(word1), len(word2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        for i in range(m + 1):
            dp[i][0] = i                 # 全删
        for j in range(n + 1):
            dp[0][j] = j                 # 全插
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if word1[i - 1] == word2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1]
                else:
                    dp[i][j] = 1 + min(dp[i - 1][j - 1],   # 替换
                                       dp[i - 1][j],       # 删除
                                       dp[i][j - 1])       # 插入
        return dp[m][n]
# 时间复杂度 O(mn)，空间 O(mn)。

###0005
# 思路：中心扩展：枚举 2n-1 个中心（单字符 / 字符间隙），
# 向两侧扩展直到不匹配，记录最长回文。
class Solution:
    def longestPalindrome(self, s: str) -> str:
        def expand(l, r):
            while l >= 0 and r < len(s) and s[l] == s[r]:
                l -= 1
                r += 1
            return l + 1, r - 1          # 返回回文的实际左右端点

        start, end = 0, 0
        for i in range(len(s)):
            l1, r1 = expand(i, i)        # 奇数长度
            l2, r2 = expand(i, i + 1)    # 偶数长度
            if r1 - l1 > end - start:
                start, end = l1, r1
            if r2 - l2 > end - start:
                start, end = l2, r2
        return s[start:end + 1]
# 时间复杂度 O(n^2)，空间 O(1)。

###0169
# 思路：Boyer-Moore 投票：同阵营 +1、异阵营 -1，归零换候选；
# 多数元素超过一半，最终存活者即答案。
class Solution:
    def majorityElement(self, nums: List[int]) -> int:
        candidate, count = 0, 0
        for x in nums:
            if count == 0:
                candidate = x            # 归零换候选
                count = 1
            elif x == candidate:
                count += 1
            else:
                count -= 1
        return candidate
# 时间复杂度 O(n)，空间 O(1)。

###0075
# 思路：荷兰国旗三指针：[0,zero) 全 0，(two, n-1] 全 2；
# 遇 2 换来的是未检查元素，cur 不前进。
class Solution:
    def sortColors(self, nums: List[int]) -> None:
        """原地修改。"""
        zero, two, cur = 0, len(nums) - 1, 0
        while cur <= two:
            if nums[cur] == 0:
                nums[cur], nums[zero] = nums[zero], nums[cur]
                zero += 1
                cur += 1
            elif nums[cur] == 2:
                nums[cur], nums[two] = nums[two], nums[cur]
                two -= 1                 # cur 不动：换来的还没检查
            else:
                cur += 1
# 时间复杂度 O(n)，空间 O(1)。

###0031
# 思路：字典序下一个排列：从右找第一个升序对 (i,i+1)，
# 再从右找第一个大于 nums[i] 的 j 交换，最后反转 i 之后的降序段。
class Solution:
    def nextPermutation(self, nums: List[int]) -> None:
        """原地修改。"""
        n = len(nums)
        i = n - 2
        while i >= 0 and nums[i] >= nums[i + 1]:   # 找第一个升序位置
            i -= 1
        if i >= 0:
            j = n - 1
            while nums[j] <= nums[i]:              # 第一个大于 nums[i] 的数
                j -= 1
            nums[i], nums[j] = nums[j], nums[i]    # 交换
        l, r = i + 1, n - 1
        while l < r:                               # 反转尾部（降序 -> 升序）
            nums[l], nums[r] = nums[r], nums[l]
            l += 1
            r -= 1
# 时间复杂度 O(n)，空间 O(1)。

###0287
# 思路：Floyd 判圈（把数组当链表 i -> nums[i]）：有 n+1 个数取值 1..n，
# 必有环，重复数字即环入口。两阶段：环内相遇 + 从起点找入口。
class Solution:
    def findDuplicate(self, nums: List[int]) -> int:
        slow = fast = nums[0]
        while True:                      # 阶段一：环内相遇
            slow = nums[slow]            # 一步
            fast = nums[nums[fast]]      # 两步
            if slow == fast:
                break
        slow = 0                         # 阶段二：找入口
        while slow != fast:
            slow = nums[slow]
            fast = nums[fast]
        return slow
# 时间复杂度 O(n)，空间 O(1)，不修改数组。
