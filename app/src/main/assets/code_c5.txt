###0070
// 思路：斐波那契递推：f(n) = f(n-1) + f(n-2)，滚动变量 O(1) 空间。
int climbStairs(int n) {
    int prev = 1, cur = 1;                   // f(0) = f(1) = 1
    for (int i = 2; i <= n; i++) {
        int next = prev + cur;
        prev = cur;
        cur = next;
    }
    return cur;
}
// 时间复杂度 O(n)，空间 O(1)。

###0118
// 思路：逐行构造：每行首尾为 1，中间 = 上一行相邻两数之和。
#include <stdlib.h>

int** generate(int numRows, int* returnSize, int** returnColumnSizes) {
    int** res = malloc(sizeof(int*) * numRows);
    *returnColumnSizes = malloc(sizeof(int) * numRows);
    *returnSize = numRows;
    for (int i = 0; i < numRows; i++) {
        res[i] = malloc(sizeof(int) * (i + 1));
        (*returnColumnSizes)[i] = i + 1;
        for (int j = 0; j <= i; j++) {
            if (j == 0 || j == i) res[i][j] = 1;
            else res[i][j] = res[i - 1][j - 1] + res[i - 1][j];
        }
    }
    return res;
}
// 时间复杂度 O(n^2)，空间 O(1)（不计输出）。

###0198
// 思路：打家劫舍 DP：f(i) = max(f(i-1) 不偷, f(i-2) + nums[i] 偷)，滚动变量。
int rob(int* nums, int numsSize) {
    int prev2 = 0, prev1 = 0;                // f(i-2), f(i-1)
    for (int i = 0; i < numsSize; i++) {
        int cur = prev1 > prev2 + nums[i] ? prev1 : prev2 + nums[i];
        prev2 = prev1;
        prev1 = cur;
    }
    return prev1;
}
// 时间复杂度 O(n)，空间 O(1)。

###0279
// 思路：完全背包：f(i) = min(f(i - j*j) + 1)，j*j <= i。
#include <stdlib.h>

int numSquares(int n) {
    int* dp = malloc(sizeof(int) * (n + 1));
    for (int i = 1; i <= n; i++) dp[i] = 0x3f3f3f3f;   // 无穷大
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j * j <= i; j++) {   // 枚举最后一个平方数
            if (dp[i - j * j] + 1 < dp[i]) dp[i] = dp[i - j * j] + 1;
        }
    }
    return dp[n];
}
// 时间复杂度 O(n·sqrt(n))，空间 O(n)。

###0322
// 思路：完全背包最少硬币数：f(a) = min(f(a - coin) + 1)，
// 初始 f(0)=0、其余无穷；结果仍为无穷表示凑不出。
#include <stdlib.h>

int coinChange(int* coins, int coinsSize, int amount) {
    int INF = amount + 1;                    // "凑不出"的哨兵值
    int* dp = malloc(sizeof(int) * (amount + 1));
    for (int a = 0; a <= amount; a++) dp[a] = INF;
    dp[0] = 0;
    for (int a = 1; a <= amount; a++) {
        for (int c = 0; c < coinsSize; c++) {
            if (coins[c] <= a && dp[a - coins[c]] + 1 < dp[a])
                dp[a] = dp[a - coins[c]] + 1;
        }
    }
    return dp[amount] > amount ? -1 : dp[amount];
}
// 时间复杂度 O(amount × 硬币数)，空间 O(amount)。

###0139
// 思路：DP：dp[i] 表示 s 前 i 个字符能否被拆分；枚举"最后一个单词"匹配。
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

bool wordBreak(char* s, char** wordDict, int wordDictSize) {
    int n = strlen(s);
    bool* dp = calloc(n + 1, sizeof(bool));
    dp[0] = true;                            // 空串可拆分
    for (int i = 1; i <= n; i++) {
        for (int w = 0; w < wordDictSize; w++) {
            int len = strlen(wordDict[w]);
            // s[i-len..i) 与单词相等，且前缀 s[0..i-len) 可拆
            if (len <= i && dp[i - len] && strncmp(s + i - len, wordDict[w], len) == 0) {
                dp[i] = true;
                break;
            }
        }
    }
    return dp[n];
}
// 时间复杂度 O(n^2·L)（上界），空间 O(n)。

###0300
// 思路：贪心 + 二分：tails[i] = 长度 i+1 的递增子序列的最小尾元素。
// 对每个数二分它应放的位置（第一个 >= 它的槽位），O(n log n)。
#include <stdlib.h>

static int lowerBound(int* tails, int size, int target) {
    int lo = 0, hi = size;
    while (lo < hi) {
        int mid = (lo + hi) / 2;
        if (tails[mid] >= target) hi = mid;
        else lo = mid + 1;
    }
    return lo;
}

int lengthOfLIS(int* nums, int numsSize) {
    int* tails = malloc(sizeof(int) * numsSize);
    int size = 0;
    for (int i = 0; i < numsSize; i++) {
        int pos = lowerBound(tails, size, nums[i]);
        tails[pos] = nums[i];                // 替换或延长
        if (pos == size) size++;
    }
    return size;
}
// 时间复杂度 O(n log n)，空间 O(n)。

###0152
// 思路：同时维护以 i 结尾的最大/最小乘积（负数会翻转极值），与 Java 版一致。
int maxProduct(int* nums, int numsSize) {
    int fmax = nums[0], fmin = nums[0], best = nums[0];
    for (int i = 1; i < numsSize; i++) {
        int x = nums[i];
        int candMax = x;
        if ((long)fmax * x > candMax) candMax = fmax * x;
        if ((long)fmin * x > candMax) candMax = fmin * x;
        int candMin = x;
        if ((long)fmax * x < candMin) candMin = fmax * x;
        if ((long)fmin * x < candMin) candMin = fmin * x;
        fmax = candMax;
        fmin = candMin;
        if (fmax > best) best = fmax;
    }
    return best;
}
// 时间复杂度 O(n)，空间 O(1)。

###0416
// 思路：01 背包可行性：目标 sum/2；dp[t] = 能否凑出和 t，
// 每个数只用一次，t 从大到小遍历。
#include <stdbool.h>
#include <stdlib.h>

bool canPartition(int* nums, int numsSize) {
    int sum = 0;
    for (int i = 0; i < numsSize; i++) sum += nums[i];
    if (sum % 2 != 0) return false;          // 总和为奇数，不可能平分
    int target = sum / 2;
    bool* dp = calloc(target + 1, sizeof(bool));
    dp[0] = true;
    for (int i = 0; i < numsSize; i++) {
        int x = nums[i];
        for (int t = target; t >= x; t--) {  // 倒序：每个数只用一次
            if (dp[t - x]) dp[t] = true;
        }
        if (dp[target]) return true;         // 提前结束
    }
    return dp[target];
}
// 时间复杂度 O(n·target)，空间 O(target)。

###0032
// 思路：栈存"最后一个未匹配位置"（哨兵）。
// '(' 下标入栈；')' 弹栈：栈空则当前下标成为新哨兵，否则长度 = i - 新栈顶。
#include <stdlib.h>

int longestValidParentheses(char* s) {
    int* stack = malloc(sizeof(int) * (strlen(s) + 1));
    int top = 0, best = 0;
    stack[top++] = -1;                       // 哨兵
    for (int i = 0; s[i]; i++) {
        if (s[i] == '(') {
            stack[top++] = i;
        } else {
            top--;                           // 弹出 '(' 或旧哨兵
            if (top == 0) {
                stack[top++] = i;            // 无配对，成为新哨兵
            } else {
                if (i - stack[top - 1] > best) best = i - stack[top - 1];
            }
        }
    }
    return best;
}
// 时间复杂度 O(n)，空间 O(n)。

###0062
// 思路：一维滚动 DP：dp[j] += dp[j-1]（旧 dp[j] 是上方，dp[j-1] 是左方）。
#include <stdlib.h>

int uniquePaths(int m, int n) {
    int* dp = malloc(sizeof(int) * n);
    for (int j = 0; j < n; j++) dp[j] = 1;   // 第一行全为 1
    for (int i = 1; i < m; i++)
        for (int j = 1; j < n; j++)
            dp[j] += dp[j - 1];
    return dp[n - 1];
}
// 时间复杂度 O(mn)，空间 O(n)。

###0064
// 思路：一维滚动 DP：f(i,j) = grid[i][j] + min(上, 左)。
// 每行开始先更新 dp[0]（第一列只能从上来）。
#include <stdlib.h>

int minPathSum(int** grid, int gridSize, int* gridColSize) {
    int m = gridSize, n = gridColSize[0];
    int* dp = malloc(sizeof(int) * n);
    dp[0] = grid[0][0];
    for (int j = 1; j < n; j++) dp[j] = dp[j - 1] + grid[0][j];   // 第一行
    for (int i = 1; i < m; i++) {
        dp[0] += grid[i][0];
        for (int j = 1; j < n; j++) {
            int up = dp[j], left = dp[j - 1];
            dp[j] = (up < left ? up : left) + grid[i][j];
        }
    }
    return dp[n - 1];
}
// 时间复杂度 O(mn)，空间 O(n)。

###1143
// 思路：LCS 二维 DP：尾字符相等取对角线+1，否则取两侧最优。
// malloc 一张 (m+1)×(n+1) 的表。
#include <stdlib.h>
#include <string.h>

int longestCommonSubsequence(char* text1, char* text2) {
    int m = strlen(text1), n = strlen(text2);
    int** dp = malloc(sizeof(int*) * (m + 1));
    for (int i = 0; i <= m; i++) dp[i] = calloc(n + 1, sizeof(int));
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (text1[i - 1] == text2[j - 1]) dp[i][j] = dp[i - 1][j - 1] + 1;
            else dp[i][j] = dp[i - 1][j] > dp[i][j - 1] ? dp[i - 1][j] : dp[i][j - 1];
        }
    }
    return dp[m][n];
}
// 时间复杂度 O(mn)，空间 O(mn)。

###0072
// 思路：编辑距离 DP：尾字符相同无操作；否则 = min(替换, 删除, 插入) + 1。
// 边界：f(i,0)=i（全删），f(0,j)=j（全插）。
#include <stdlib.h>
#include <string.h>

int minDistance(char* word1, char* word2) {
    int m = strlen(word1), n = strlen(word2);
    int** dp = malloc(sizeof(int*) * (m + 1));
    for (int i = 0; i <= m; i++) dp[i] = malloc(sizeof(int) * (n + 1));
    for (int i = 0; i <= m; i++) dp[i][0] = i;
    for (int j = 0; j <= n; j++) dp[0][j] = j;
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (word1[i - 1] == word2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1];
            } else {
                int best = dp[i - 1][j - 1];                    // 替换
                if (dp[i - 1][j] < best) best = dp[i - 1][j];   // 删除
                if (dp[i][j - 1] < best) best = dp[i][j - 1];   // 插入
                dp[i][j] = best + 1;
            }
        }
    }
    return dp[m][n];
}
// 时间复杂度 O(mn)，空间 O(mn)。

###0005
// 思路：中心扩展：枚举 2n-1 个中心（单字符 / 字符间隙），
// 向两侧扩展直到不匹配，记录最长回文的起点和长度。
#include <stdlib.h>
#include <string.h>

static int expand(char* s, int l, int r) {
    while (l >= 0 && s[r] && s[l] == s[r]) { // 多走一步后退出
        l--;
        r++;
    }
    return r - l - 1;                        // 修正多走的一步
}

char* longestPalindrome(char* s) {
    int n = strlen(s);
    if (n == 0) {
        char* e = malloc(1);
        e[0] = '\0';
        return e;
    }
    int start = 0, maxLen = 0;
    for (int i = 0; i < n; i++) {
        int len1 = expand(s, i, i);          // 奇数长度
        int len2 = expand(s, i, i + 1);      // 偶数长度
        int len = len1 > len2 ? len1 : len2;
        if (len > maxLen) {
            maxLen = len;
            start = i - (len - 1) / 2;
        }
    }
    char* res = malloc(maxLen + 1);
    memcpy(res, s + start, maxLen);
    res[maxLen] = '\0';
    return res;
}
// 时间复杂度 O(n^2)，空间 O(1)（不计输出）。

###0136
// 思路：异或：a^a=0、a^0=a，成对的数全部抵消。
int singleNumber(int* nums, int numsSize) {
    int result = 0;
    for (int i = 0; i < numsSize; i++) result ^= nums[i];
    return result;
}
// 时间复杂度 O(n)，空间 O(1)。

###0169
// 思路：Boyer-Moore 投票：同阵营 +1、异阵营 -1，归零换候选；
// 多数元素超过一半，最终存活者即答案。
int majorityElement(int* nums, int numsSize) {
    int candidate = 0, count = 0;
    for (int i = 0; i < numsSize; i++) {
        if (count == 0) {
            candidate = nums[i];
            count = 1;
        } else if (nums[i] == candidate) {
            count++;
        } else {
            count--;
        }
    }
    return candidate;
}
// 时间复杂度 O(n)，空间 O(1)。

###0075
// 思路：荷兰国旗三指针：[0,zero) 全 0，(two, n-1] 全 2；
// 遇 2 换来的是未检查元素，cur 不前进。
void sortColors(int* nums, int numsSize) {
    int zero = 0, two = numsSize - 1, cur = 0;
    while (cur <= two) {
        if (nums[cur] == 0) {
            int t = nums[cur]; nums[cur] = nums[zero]; nums[zero] = t;
            zero++;
            cur++;
        } else if (nums[cur] == 2) {
            int t = nums[cur]; nums[cur] = nums[two]; nums[two] = t;
            two--;
            // cur 不动：换来的元素还没检查
        } else {
            cur++;
        }
    }
}
// 时间复杂度 O(n)，空间 O(1)。

###0031
// 思路：字典序下一个排列：从右找第一个升序对 (i,i+1)，
// 再从右找第一个大于 nums[i] 的 j 交换，最后反转 i 之后的降序段。
static void swapInt(int* a, int* b) {
    int t = *a;
    *a = *b;
    *b = t;
}

void nextPermutation(int* nums, int numsSize) {
    int i = numsSize - 2;
    while (i >= 0 && nums[i] >= nums[i + 1]) i--;    // 找第一个升序位置
    if (i >= 0) {
        int j = numsSize - 1;
        while (nums[j] <= nums[i]) j--;              // 第一个大于 nums[i] 的数
        swapInt(&nums[i], &nums[j]);
    }
    for (int l = i + 1, r = numsSize - 1; l < r; l++, r--)   // 反转尾部
        swapInt(&nums[l], &nums[r]);
}
// 时间复杂度 O(n)，空间 O(1)。

###0287
// 思路：Floyd 判圈（把数组当链表 i -> nums[i]）：
// 有 n+1 个数取值 1..n，必有环，重复数字即环入口。
// 阶段一快慢指针相遇，阶段二从起点同速前进找入口。
int findDuplicate(int* nums, int numsSize) {
    int slow = nums[0];
    int fast = nums[nums[0]];
    while (slow != fast) {                   // 阶段一：环内相遇
        slow = nums[slow];                   // 一步
        fast = nums[nums[fast]];             // 两步
    }
    slow = 0;                                // 阶段二：找入口
    while (slow != fast) {
        slow = nums[slow];
        fast = nums[fast];
    }
    return slow;
}
// 时间复杂度 O(n)，空间 O(1)，不修改数组。
